Integration Recipe (Copy & Paste Reference)

How to verify Inavate webhook signatures in your own applications:

package main

import (
    "crypto/hmac"
    "crypto/sha256"
    "crypto/subtle"
    "encoding/hex"
    "io"
    "math"
    "net/http"
    "strconv"
    "time"
)

func handleWebhook(w http.ResponseWriter, r *http.Request) {
    // 1. CRITICAL: Capture the exact RAW body bytes before any JSON parsing
    rawBody, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "Failed to read body", http.StatusBadRequest)
        return
    }

    timestamp := r.Header.Get("X-Inavate-Timestamp")
    signature := r.Header.Get("X-Inavate-Signature")

    // 2. Replay protection: Verify timestamp freshness (drift <= 300 seconds)
    tsInt, _ := strconv.ParseInt(timestamp, 10, 64)
    if math.Abs(float64(time.Now().Unix() - tsInt)) > 300 {
        http.Error(w, "Timestamp expired", http.StatusBadRequest)
        return
    }

    // 3. Compute expected HMAC-SHA256 signature
    sigPayload := timestamp + "." + string(rawBody)
    mac := hmac.New(sha256.New, []byte("YOUR_WEBHOOK_SECRET"))
    mac.Write([]byte(sigPayload))
    expectedSig := hex.EncodeToString(mac.Sum(nil))

    // 4. Constant-time comparison to prevent timing attacks
    if subtle.ConstantTimeCompare([]byte(signature), []byte(expectedSig)) != 1 {
        http.Error(w, "Invalid signature", http.StatusUnauthorized)
        return
    }

    // 5. Acknowledge with 200 OK immediately
    w.WriteHeader(http.StatusOK)
    w.Write([]byte(`{"status":"acknowledged"}`))
}
const express = require('express');
const crypto = require('crypto');
const app = express();

// 1. CRITICAL: Capture raw body buffer via express.raw
app.post('/webhook', express.raw({ type: 'application/json' }), (req, res) => {
    const rawBody = req.body.toString('utf8');
    const timestamp = req.headers['x-inavate-timestamp'];
    const signature = req.headers['x-inavate-signature'];

    // 2. Replay check (within 300s)
    const now = Math.floor(Date.now() / 1000);
    if (Math.abs(now - parseInt(timestamp, 10)) > 300) {
        return res.status(400).send('Timestamp expired');
    }

    // 3. Compute HMAC-SHA256
    const sigPayload = `${timestamp}.${rawBody}`;
    const expectedSig = crypto
        .createHmac('sha256', process.env.WEBHOOK_SECRET)
        .update(sigPayload)
        .digest('hex');

    // 4. Constant-time comparison
    const isValid = crypto.timingSafeEqual(
        Buffer.from(signature, 'hex'),
        Buffer.from(expectedSig, 'hex')
    );
    if (!isValid) return res.status(401).send('Invalid signature');

    // 5. Return 200 OK
    res.status(200).json({ status: 'acknowledged' });
});
import hmac, hashlib, time
from flask import Flask, request, jsonify

app = Flask(__name__)

# 1. Capture raw request bytes
@app.route('/webhook', methods=['POST'])
def webhook():
    raw_body = request.get_data(as_text=True)
    timestamp = request.headers.get('X-Inavate-Timestamp')
    signature = request.headers.get('X-Inavate-Signature')

    # 2. Replay check (within 300s)
    if abs(time.time() - int(timestamp)) > 300:
        return "Timestamp expired", 400

    # 3. Compute HMAC-SHA256
    sig_payload = f"{timestamp}.{raw_body}"
    expected_sig = hmac.new(
        b"YOUR_WEBHOOK_SECRET",
        sig_payload.encode('utf-8'),
        hashlib.sha256
    ).hexdigest()

    # 4. Constant-time compare
    if not hmac.compare_digest(signature, expected_sig):
        return "Invalid signature", 401

    # 5. Return 200 OK
    return jsonify({"status": "acknowledged"}), 200
